# Python dependencies for the data-prep, weather, resource, and load stages.
# Install with:  pip install -r requirements.txt
#
# The dispatch stage (4_dispatch/Syngrid2025.m) is MATLAB and is NOT covered here.

pandas
numpy
scipy

pvlib                # solar PV modeling
requests             # NASA POWER API for wind IEC classification
